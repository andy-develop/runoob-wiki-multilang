クイックソート

より後ろの数はすべて大きいことがわかります。したがって、a[0…4]とa[6…9]の2つの部分区間に対して上記の手順を繰り返せばよいのです。 プログラミング技術

クイックソートは、同じO(N*logN)のソート方法の中でも効率が高いため、よく採用されます。さらに、クイックソートの考え方である分割統治法は実際に実用的です。そのため、多くのソフトウェア企業の筆記試験や面接（テンセントやマイクロソフトなどの有名IT企業を含む）で好んで出題され、また大小様々なプログラミング関連の試験（軟考や大学院入試など）でもクイックソートがよく登場します。

=a

 

クイックソートは、C.R.A.Hoareが1962年に提案した分割交換ソートです。分割統治戦略を採用しており、一般に分割統治法（Divide-and-ConquerMethod）と呼ばれます。

この方法の基本的な考え方は次のとおりです。

  • 1. まず数列から1つの数を取り出し、基準数とします。
  • 2. 分割処理では、この数より大きい数をすべて右側に置き、この数以下の数をすべて左側に置きます。
  • 3. 左右の区間に対して2番目のステップを、各区間が1つの数だけになるまで繰り返します。

      • クイックソートは分割統治法と呼ばれますが、「分割統治法」の3文字だけではクイックソートの全手順をうまく説明できません。そこで、クイックソートについてさらに詳しく説明します。穴掘り・数埋め法＋分割統治法です。

      まず実例を見てみましょう。定義は後で示します（定義を自分の言葉でまとめると、コードの実装に役立ちます）。

      穴掘り・数埋め法をまとめると：

      • 1. i = L; j = R; 基準数を掘り出して最初の穴a[i]を作ります。
      • 2. j-- で後ろから前へそれより小さい数を探し、見つけたらその数を掘り出して前の穴a[i]に埋めます。
      • 3. i++ で前から後ろへそれより大きい数を探し、見つけたらその数を掘り出して前の穴a[j]に埋めます。
      • 4. i==jになるまで2と3の手順を繰り返し、基準数をa[i]に埋めます。

      このまとめに従えば、穴掘り・数埋めのコードを簡単に実装できます：

      int AdjustArray(int s[], int l, int r) //調整後の基準数の位置を返す { int i = l, j = r; int x = s[l]; //s[l]すなわちs[i]が最初の穴です while (i < j) { // 右から左へxより小さい数を探してs[i]を埋める while(i < j && s[j] >= x) j--; if(i < j) { s[i] = s[j]; //s[j]をs[i]に埋めると、s[j]が新しい穴になります i++; } // 左から右へx以上の数を探してs[j]を埋める while(i < j && s[i] < x) i++; if(i < j) { s[j] = s[i]; //s[i]をs[j]に埋めると、s[i]が新しい穴になります j--; } } //終了時、iとjは等しい。xをこの穴に埋める。 s[i] = x; return i; }

      次に分割統治法のコードを書きます：

      void quick_sort1(int s[], int l, int r) { if (l < r) { int i = AdjustArray(s, l, r);//まず穴掘り・数埋め法でs[]を調整する quick_sort1(s, l, i - 1); //再帰呼び出し quick_sort1(s, i + 1, r); } }

      このコードは明らかに簡潔ではないので、整理してまとめます：

      //クイックソート void quick_sort(int s[], int l, int r) { if (l < r) { //Swap(s[l], s[(l + r) / 2]); //中央の数と最初の数を交換する 注1を参照 int i = l, j = r, x = s[l]; while (i < j) { while(i < j && s[j] >= x) // 右から左へxより小さい最初の数を探す j--; if(i < j) s[i++] = s[j]; while(i < j && s[i] < x) // 左から右へx以上の最初の数を探す i++; if(i < j) s[j--] = s[i]; } s[i] = x; quick_sort(s, l, i - 1); //再帰呼び出し quick_sort(s, i + 1, r); } }

      クイックソートには多くの改良版があります。例えば、基準数をランダムに選んだり、区間内のデータが少ない場合は別のソート方法を直接使って再帰の深さを減らしたりします。興味がある方はさらに深く研究してみてください。

      注1：一部の書籍では中央の数を基準数にしています。これを実装するのはとても簡単で、中央の数と最初の数を交換するだけです。

    namespace{
    class Program{
        static void QuickSort(int[] dataArray,int left,int right){
            if(left < right){
                int x=dataArray;
                int i=left;
                int j=right;
                while(true && i<j)
                {
                    while(true && i<j){
                        if(dataArray[j]<=x){
                            dataArray[i]=dataArray[j];
                            break;
                        }else{
                            j--;
                        }
                    }
                    while(true && i<j){
                        if(dataArray[i]>x){
                            dataArray[j]=dataArray[i];
                            break;          
                        }//if结束
                        else{
                            i++;
                        }
                    }//while结束
                }//第一个While结束

                //跳出循环，现在i==j了，i是中间位。
                dataArray[i]=x;
                QuickSort(dataArray,left, i-1);
                QuickSort(dataArray,i+1,right);
            }
        }
        static void Main(string[] args){
            int[] data=new int[]{72,6,57,88,60,42,83,73,48,85};
            Console.WriteLine("快速排序的顺序为：");
            QuickSort(data,0,data.Length-1);
            foreach(var item in data){
                Console.Write(item + " ");
            }
            Console.ReadLine();
        }
    }
}

public class Quick {
    public static void main(String[] args) {
        int[] arr={8,4,5,7,1,3,6};//直接复制数组
        quick_sort(arr,0,arr.length-1);    
        print(arr);

    private static int get_mid(int arr[],int left,int right){
        int pivot=arr[left];//自定义排序中心轴，这里把arr[left]存到pivot中去，此时arr[left]为空。pivot相当于一个中间量
        while(left<right){//当left与right指针相遇的时候退出循环，双指针遍历结束
            while(arr[right]>=pivot && left<right) right--;//right指针从右往左遍历，当arr[right]>=pivot，即满足以pivot为中轴，小放左，大放右的条件时，right指针继续往右遍历。当arr[right]<pivotd的时候，把当前值arr[right]赋给空置arr[left]，此时arr[right]成了空值。
                 arr[left]=arr[right];
            while(arr[left]<=pivot && left<right) left++;//到left指针从左往右遍历，当arr[left]<=pivot，即满足以pivot为中轴，小放左，大放右的条件时，left指针继续往左遍历。当arr[left]>pivot的时候，把当前值arr[left]赋给空置arr[right]，此时arr[left]成了空值。
                 arr[right]=arr[left];
        }
        //经历了上面的循环实现了pivot为中轴，小放左，大放右的格局
        arr[left]=pivot;//最后把存放在pivot值放回数组空arr[left]中
        return left;//返回中轴所在的下标位置。
    }
    
    private static void  quick_sort(int[] arr,int left,int right){
        if(left<right){
       /*将arr[left..right]均分为两部分arr[left..mid]和arr[mid+1..right]
        * ,以pivot为中轴，小放左，大放右。这是第一步。*/
            int mid =get_mid(arr,left,right);//接收中轴所在的下标位置。
            quick_sort(arr,left,mid-1);//递归地对arr[left..mid]进行快速排序，使得左子序列有序
            quick_sort(arr,mid+1,right);//递归地对arr[mid+1..right]进行快速排序，使得左子序列有序
        }
    }
    
    public static void print(int arr[])//封装函打印函数
    {
        for(int k=0;k<arr.length;k++)
          {
            System.out.print(arr[k]+" ");
          }
    }
}

    <?php
function swap(array &$arr,$a,$b){
    $temp = $arr[$a];
    $arr[$a] = $arr[$b];
    $arr[$b] = $temp;
}

//基准：选取数组当中的一个关键字，使得它处于数组某个位置时，左边的值比它小，右边的值比它大
function getKey(&$arr, $low, $high){
    $key = $arr[$low];                                    //选取子数组第一个元素作为枢轴，因为key是从低端取的，所以先要从高端开始往前找，否则相反
    while($low < $high){                                 //从数组的两端交替向中间扫描（当 $low = $high 结束循环）
        while($low < $high && $arr[$high] >= $key){      //保证high下标以后的元素都大于基准
            $high--;         
        }
        swap($arr,$low,$high);                           //将比基准小的元素交换到低端
        while($low < $high && $arr[$low] <= $key){       //保证low下标以前的元素都小于基准
            $low++;
        }
        swap($arr,$low,$high);                          //将比基准大的元素交换到高端
    }
    return $low;
}
function QSort(&$arr, $low, $high){
    if($low < $high){
        $key = getKey($arr,$low,$high);     //找基准，使得基准左边的元素<=基准，右边的元素>=基准
        QSort($arr, $low, $key-1);          //对低子表($key左边的记录)递归排序
        QSort($arr, $key+1, $high);         //对高子表($key右边的记录)递归排序
    }
}
function QuickSort(&$arr){
    $low = 0;
    $high = count($arr) -1;
    QSort($arr, $low, $high);
}

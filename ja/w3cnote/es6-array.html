3.2.4 ES6 配列

分類 ES6 チュートリアル

配列の作成

Array.of()

引数のすべての値を要素として配列を生成します。

console.log(Array.of(1, 2, 3, 4)); // [1, 2, 3, 4] //引数の値は異なる型でも構いません。 console.log(Array.of(1, '2', true)); // [1, '2', true] //引数が空の場合、空の配列を返します。 console.log(Array.of()); // []

Array.from()

配列ライクオブジェクトまたはイテラブルオブジェクトを配列に変換します。

//引数が配列の場合、元の配列と同じ配列を返します。 console.log(Array.from([1, 2])); // [1, 2] //引数に空の要素が含まれる場合 console.log(Array.from([1, , 3])); // [1, undefined, 3]

引数

Array.from(arrayLike[, mapFn[, thisArg]])

戻り値は変換後の配列です。

arrayLike

変換したい配列ライクオブジェクトまたはイテラブルオブジェクト。

console.log(Array.from([1, 2, 3])); // [1, 2, 3]

mapFn

省略可能。map 関数。各要素を処理し、処理後の要素を配列に入れます。

console.log(Array.from([1, 2, 3], (n) => n * 2)); // [2, 4, 6]

thisArg

省略可能。map 関数実行時の this オブジェクトを指定します。

let map = { do: function(n) { return n * 2; } } let arrayLike = [1, 2, 3]; console.log(Array.from(arrayLike, function (n){ return this.do(n); }, map)); // [2, 4, 6]

配列ライクオブジェクト

配列ライクオブジェクトは length プロパティを持ち、要素のプロパティ名は数値または数値に変換可能な文字列である必要があります。

let arr = Array.from({ 0: '1', 1: '2', 2: 3, length: 3 }); console.log(arr); // ['1', '2', 3] //length プロパティがない場合、空の配列を返します。 let array = Array.from({ 0: '1', 1: '2', 2: 3, }); console.log(array); // [] //要素のプロパティ名が数値でなく数値に変換できない場合、長さが length で要素の値が undefined の配列を返します。 let array1 = Array.from({ a: 1, b: 2, length: 2 }); console.log(array1); // [undefined, undefined]

イテラブルオブジェクトの変換

Map の変換

let map = new Map(); map.set('key0', 'value0'); map.set('key1', 'value1'); console.log(Array.from(map)); // [['key0', 'value0'],['key1', // 'value1']]

Set の変換

let arr = [1, 2, 3]; let set = new Set(arr); console.log(Array.from(set)); // [1, 2, 3]

文字列の変換

let str = 'abc'; console.log(Array.from(str)); // ["a", "b", "c"]

拡張メソッド

検索

find()

配列内で条件に一致する要素を検索します。複数の一致する要素がある場合、最初の要素を返します。

let arr = Array.of(1, 2, 3, 4); console.log(arr.find(item => item > 2)); // 3 //配列の空の要素は undefined として扱われます。 console.log([, 1].find(n => true)); // undefined

findIndex()

配列内で条件に一致する要素のインデックスを検索します。複数の一致する要素がある場合、最初の要素のインデックスを返します。

let arr = Array.of(1, 2, 1, 3); //引数1: コールバック関数 //引数2(省略可能): コールバック関数内の this の値を指定 console.log(arr.findIndex(item => item == 2)); // 1 //配列の空の要素は undefined として扱われます。 console.log([, 1].findIndex(n => true)); //0

埋め込み

fill()

一定のインデックス範囲の配列要素を、単一の指定された値で埋めます。

let arr = Array.of(1, 2, 3, 4); //引数1: 埋めるために使用する値 //引数2: 埋め込みを開始するインデックス //引数3(省略可能): 埋め込みの終了インデックス。デフォルトは配列の末尾 console.log(arr.fill(0,1,2)); // [1, 0, 3, 4]

copyWithin()

一定のインデックス範囲の配列要素を、この配列の別の指定されたインデックス範囲の要素で変更します。

//引数1: 変更対象の開始インデックス //引数2: 上書きに使用するデータの開始インデックス //引数3(省略可能): 上書きに使用するデータの終了インデックス。デフォルトは配列の末尾 console.log([1, 2, 3, 4].copyWithin(0,2,4)); // [3, 4, 3, 4] //引数1が負の数の場合、末尾から数えた位置を表します。 console.log([1, 2, 3, 4].copyWithin(-2, 0)); // [1, 2, 1, 2] console.log([1, 2, ,4].copyWithin(0, 2, 4)); // [, 4, , 4]

反復

entries()

キーと値のペアを反復処理します。

for(let [key, value] of ['a', 'b'].entries()){ console.log(key, value); } // 0 "a" // 1 "b" //for... of ループを使用しません。 let entries = ['a', 'b'].entries(); console.log(entries.next().value); // [0, "a"] console.log(entries.next().value); // [1, "b"] //配列に空の要素が含まれる場合 console.log([...[,'a'].entries()]); // [[0, undefined], [1, "a"]]

keys()

キー名を反復処理します。

for(let key of ['a', 'b'].keys()){ console.log(key); } // 0 // 1 //配列に空の要素が含まれる場合 console.log([...[,'a'].keys()]); // [0, 1]

values()

キーの値を反復処理します。

for(let value of ['a', 'b'].values()){ console.log(value); } // "a" // "b" //配列に空の要素が含まれる場合 console.log([...[,'a'].values()]); // [undefined, "a"]

包含

includes()

配列が指定された値を含むかどうか。

注意: Set と Map の has メソッドとは区別してください。Set の has メソッドは値の検索に使用され、Map の has メソッドはキー名の検索に使用されます。

//引数1: 含まれるか確認する指定値 [1, 2, 3].includes(1); // true //引数2: 省略可能、検索の開始インデックス。デフォルトは0 [1, 2, 3].includes(1, 2); // false //NaN の包含判定 [1, NaN, 3].includes(NaN); // true

ネストした配列を1次元配列に変換

flat()

console.log([1 ,[2, 3]].flat()); // [1, 2, 3] //変換するネストの深さを指定 console.log([1, [2, [3, [4, 5]]]].flat(2)); // [1, 2, 3, [4, 5]] //ネストの深さに関係なく console.log([1, [2, [3, [4, 5]]]].flat(Infinity)); // [1, 2, 3, 4, 5] //空の要素は自動的にスキップされます。 console.log([1, [2, , 3]].flat());<p> // [1, 2, 3]
flatMap()

配列内の各要素に処理を行ってから、配列に flat() メソッドを実行します。

//引数1: 反復関数。この反復関数は3つの引数を受け取れます: 現在の要素、現在の要素のインデックス、元の配列 //引数2: 反復関数内の this の参照先を指定 console.log([1, 2, 3].flatMap(n => [n * 2])); // [2, 4, 6]

配列バッファ

配列バッファはメモリ内の連続したアドレス領域です。

型付き配列の基盤です。

実際のバイト数は作成時に決定され、その後はデータを変更できるだけで、サイズは変更できません。

配列バッファの作成

コンストラクタを使用して作成:

let buffer = new ArrayBuffer(10); console.log(buffer.byteLength); // 10既存の配列バッファの分割let buffer = new ArrayBuffer(10); let buffer1 = buffer.slice(1, 3); console.log(buffer1.byteLength); // 2

ビュー

ビューはメモリを操作するためのインターフェースです。

ビューは配列バッファまたはバッファのバイトのサブセットを操作でき、いずれかの数値データ型に従ってデータを読み書きします。

DataView 型は汎用の配列バッファビューであり、8種類すべての数値データ型をサポートします。

作成:

//デフォルトでは DataView は配列バッファの全内容を操作できます。 let buffer = new ArrayBuffer(10); dataView = new DataView(buffer); dataView.setInt8(0,1); console.log(dataView.getInt8(0)); // 1 //オフセット(引数2)と長さ(引数3)を設定することで、DataView が操作できるバイト範囲を指定します。 let buffer1 = new ArrayBuffer(10); dataView1 = new DataView(buffer1, 0, 3); dataView1.setInt8(5,1); // RangeError

型付き配列

配列バッファの特定の型のビューです。

汎用の DataView オブジェクトを使用して配列バッファを操作する代わりに、特定のデータ型を強制的に使用できます。

作成

配列バッファから生成

let buffer = new ArrayBuffer(10), view = new Int8Array(buffer); console.log(view.byteLength); // 10

コンストラクタ経由

let view = new Int32Array(10); console.log(view.byteLength); // 40 console.log(view.length); // 10 //引数を渡さない場合、デフォルトの長さは0です。 //この場合、配列バッファに領域が割り当てられないため、作成した型付き配列はデータを保存できません。 let view1 = new Int32Array(); console.log(view1.byteLength); // 0 console.log(view1.length); // 0 //引数として、型付き配列、イテラブルオブジェクト、配列、配列ライクオブジェクトを受け取れます。 let arr = Array.from({ 0: '1', 1: '2', 2: 3, length: 3 }); let view2 = new Int16Array([1, 2]), view3 = new Int32Array(view2), view4 = new Int16Array(new Set([1, 2, 3])), view5 = new Int16Array([1, 2, 3]), view6 = new Int16Array(arr); console.log(view2 .buffer === view3.buffer); // false console.log(view4.byteLength); // 6 console.log(view5.byteLength); // 6 console.log(view6.byteLength); // 6

注意点

length プロパティは書き込み不可です。この値を変更しようとすると、非厳格モードではその操作は無視され、厳格モードではエラーがスローされます。

let view = new Int16Array([1, 2]); view.length = 3; console.log(view.length); // 2

型付き配列は entries()、keys()、values() を使用して反復処理できます。

let view = new Int16Array([1, 2]); for(let [k, v] of view.entries()){ console.log(k, v); } // 0 1 // 1 2

find() などのメソッドも型付き配列で使用できます。ただし、型付き配列のメソッドは数値型が安全かどうかを追加でチェックし、Symbol.species によってメソッドの戻り値が通常の配列ではなく型付き配列であることを確認します。concat() メソッドは、2つの型付き配列の結合結果が不確定なため、型付き配列には使用できません。また、型付き配列のサイズは変更できないため、splice() のように配列のサイズを変更できるメソッドは型付き配列には適用できません。

let view = new Int16Array([1, 2]); view.find((n) > 1); // 2

すべての型付き配列には静的メソッド of() と from() があります。実行結果はそれぞれ Array.of() メソッドと Array.from() メソッドに似ていますが、型付き配列のメソッドは型付き配列を返すのに対し、通常の配列のメソッドは通常の配列を返すという違いがあります。

let view = Int16Array.of(1, 2); console.log(view instanceof Int16Array); // true

型付き配列は通常の配列ではなく、Array を継承しません。

let view = new Int16Array([1, 2]); console.log(Array.isArray(view)); // false

型付き配列には set() メソッドと subarray() メソッドが追加されています。set() メソッドは他の配列を既存の型付き配列にコピーするために使用され、subarray() は既存の型付き配列の一部を取り出して新しい型付き配列を作成するために使用されます。

//set メソッド //引数1: 型付き配列または通常の配列 //引数2: 省略可能、オフセット。データの挿入開始位置で、デフォルトは0 let view= new Int16Array(4); view.set([1, 2]); view.set([3, 4], 2); console.log(view); // [1, 2, 3, 4] //subarray メソッド //引数1: 省略可能、開始位置 //引数2: 省略可能、終了位置(終了位置は含みません) let view= new Int16Array([1, 2, 3, 4]), subview1 = view.subarray(), subview2 = view.subarray(1), subview3 = view.subarray(1, 3); console.log(subview1); // [1, 2, 3, 4] console.log(subview2); // [2, 3, 4] console.log(subview3); // [2, 3]

スプレッド演算子

配列のコピー

let arr = [1, 2], arr1 = [...arr]; console.log(arr1); // [1, 2] //配列に空の要素が含まれる場合 let arr2 = [1, , 3], arr3 = [...arr2]; console.log(arr3); [1, undefined, 3]

配列の結合

console.log([...[1, 2],...[3, 4]]); // [1, 2, 3, 4]

AI Agent 垂直応用シナリオ

分野によって Agent に対するニーズと特徴は異なる。

分野に特化させることで、Agent はより専門的なサービスを提供できる。

コードエージェント

コードエージェントは、現在最も成功している Agent の応用分野の一つである。

コードベースを理解し、新しいコードを作成し、問題をデバッグできる。

コア能力

コード補完：コンテキストに基づいて次のコードを予測する。

コード生成：自然言語の説明からコードを生成する。

コードレビュー：コードの問題点や改善点を発見する。

バグの特定：エラーメッセージを分析し、問題の根本原因を特定する。

コードリファクタリング：コード最適化の提案を行い、実行する。

コード実装例

Text2SQL Agent の実装

class Text2SQLAgent:
    """
    Text2SQL Agent
自然言語をSQLクエリに変換する
    """
   
    def __init__(self, llm, schema):
        self.llm = llm
        # データベーススキーマ情報
        self.schema = schema
   
    def convert(self, question):
        """
自然言語の質問をSQLに変換する
:param question: 自然言語の質問
:return: SQLクエリ文
        """
        prompt = f"""
あなたはSQLの専門家です。

データベーススキーマ：
{self.schema}

ルール：
1. SELECTクエリのみ生成する（INSERT/UPDATE/DELETEはサポートしない）
2. 適切なテーブル別名を使用する
3. 必要なJOIN条件を追加する
4. 明確な列別名を使用する

質問：{question}

対応するSQLクエリを生成してください。
"""
        sql = self.llm.generate(prompt)
       
        # セキュリティチェック
        return self.sanitize(sql)
   
    def sanitize(self, sql):
        """
SQLセキュリティチェック
危険な操作が含まれていないことを確認する
        """
        # 小文字変換チェック
        sql_lower = sql.lower().strip()
       
        # SELECTのみを含むかチェック
        forbidden_keywords = [
            "insert", "update", "delete", "drop",
            "create", "alter", "truncate", "exec",
            "execute", "grant", "revoke"
        ]
       
        for keyword in forbidden_keywords:
            if keyword in sql_lower:
                raise ValueError(f"禁止キーワード: {keyword}")
       
        return sql
   
    def execute(self, question, db_connection):
        """
Text2SQLクエリを実行する
        """
        sql = self.convert(question)
        cursor = db_connection.cursor()
        cursor.execute(sql)
        results = cursor.fetchall()
        columns = [desc[0] for desc in cursor.description]
        return {"columns": columns, "rows": results}

データ分析エージェント

データ分析エージェントはデータ要件を理解し、クエリを実行し、分析レポートを生成できる。

非技術者でも複雑なデータ分析を行えるようにする。

コア能力

データ理解：データベース構造とデータの意味を理解する。

クエリ生成：Text2SQL、自然言語をクエリに変換する。

可視化：グラフの提案と設定を生成する。

レポート生成：分析結果から自然言語レポートを生成する。

コード実装

データ分析エージェントの実装

class DataAnalysisAgent:
    """
データ分析エージェント
データ要件を理解し、分析を実行し、レポートを生成する
    """
   
    def __init__(self, llm, db_connection, chart_generator):
        self.llm = llm
        self.db = db_connection
        self.chart_generator = chart_generator
   
    def analyze(self, request):
        """
データ分析リクエストを処理する
        """
        # ステップ1：分析要件を理解する
        intent = self.parse_intent(request)
       
        # ステップ2：クエリを生成する
        query = self.generate_query(intent)
       
        # ステップ3：クエリを実行する
        data = self.execute_query(query)
       
        # ステップ4：データを分析する
        analysis = self.perform_analysis(data, intent)
       
        # ステップ5：可視化の提案を生成する
        charts = self.suggest_charts(analysis)
       
        # ステップ6：レポートを生成する
        report = self.generate_report(analysis, charts)
       
        return {
            "intent": intent,
            "query": query,
            "data": data,
            "analysis": analysis,
            "charts": charts,
            "report": report
        }
   
    def parse_intent(self, request):
        """ユーザーの意図を解析する"""
        prompt = f"""
以下のデータリクエストを分析し、重要な情報を抽出してください：

リクエスト：{request}

以下を抽出してください：
1. 分析目的（比較、トレンド、分布など）
2. 関連するテーブルとフィールド
3. 時間範囲（ある場合）
4. フィルター条件（ある場合）
"""
        return self.llm.generate(prompt)
   
    def generate_query(self, intent):
        """SQLクエリを生成する"""
        prompt = f"""
以下の意図に基づいてSQLクエリを生成してください：

{intent}

データベーススキーマ：
{self.get_schema()}
"""
        return self.llm.generate(prompt)
   
    def execute_query(self, query):
        """クエリを実行する"""
        cursor = self.db.cursor()
        cursor.execute(query)
        columns = [desc[0] for desc in cursor.description]
        rows = cursor.fetchall()
        return {"columns": columns, "rows": rows}
   
    def perform_analysis(self, data, intent):
        """データ分析を実行する"""
        # 簡略化した実装
        import statistics
       
        rows = data["rows"]
        columns = data["columns"]
       
        analysis = {
            "row_count": len(rows),
            "summary": {}
        }
       
        # 数値列の統計情報を計算する
        for i, col in enumerate(columns):
            if self.is_numeric(rows, i):
                values = [row[i] for row in rows if row[i] is not None]
                if values:
                    analysis["summary"][col] = {
                        "min": min(values),
                        "max": max(values),
                        "avg": statistics.mean(values),
                        "median": statistics.median(values)
                    }
       
        return analysis
   
    def suggest_charts(self, analysis):
        """可視化グラフを推薦する"""
        suggestions = []
       
        # 分析タイプに応じてグラフを推薦する
        if "trend" in str(analysis).lower():
            suggestions.append({
                "type": "line",
                "description": "折れ線グラフ、トレンドの変化を示すのに適しています"
            })
       
        if "comparison" in str(analysis).lower():
            suggestions.append({
                "type": "bar",
                "description": "棒グラフ、異なるカテゴリを比較するのに適しています"
            })
       
        suggestions.append({
            "type": "table",
            "description": "データテーブル、詳細なデータを表示する"
        })
       
        return suggestions
   
    def generate_report(self, analysis, charts):
        """分析レポートを生成する"""
        prompt = f"""
以下の分析結果に基づいて、自然言語レポートを生成してください：

分析結果：
{analysis}

推薦グラフ：
{charts}

明確な分析レポートを生成してください。
"""
        return self.llm.generate(prompt)

カスタマーサービスエージェント

カスタマーサービスエージェントはマルチターン対話を処理し、ユーザーの意図を理解し、正確な返答を提供する必要がある。

AIが最も広く導入されているシーンの一つである。

コア能力

意図認識：ユーザーが何をしたいのかを理解する。

スロット埋め：重要な情報（時間、場所など）を抽出する。

対話状態の追跡：マルチターン対話のコンテキストを管理する。

応答生成：自然で専門的な応答を生成する。

コード実装

カスタマーサービスエージェントの実装

class CustomerServiceAgent:
    """
カスタマーサービスエージェント
ユーザーからの問い合わせを処理し、サポートを提供する
    """
   
    def __init__(self, llm, knowledge_base, dialogue_manager):
        self.llm = llm
        # ナレッジベース
        self.knowledge_base = knowledge_base
        # 対話マネージャー
        self.dialogue_manager = dialogue_manager
   
    def handle(self, user_input, session_id):
        """
ユーザーメッセージを処理する
        """
        # 対話履歴を取得する
        history = self.dialogue_manager.get_history(session_id)
       
        # ユーザーの意図を認識する
        intent = self.recognize_intent(user_input, history)
       
        # 意図に応じて処理する
        if intent.type == "greeting":
            response = self.handle_greeting()
        elif intent.type == "inquiry":
            response = self.handle_inquiry(intent)
        elif intent.type == "complaint":
            response = self.handle_complaint(intent)
        elif intent.type == "order_status":
            response = self.handle_order_status(intent)
        elif intent.type == "refund":
            response = self.handle_refund(intent)
        else:
            response = self.handle_general(user_input)
       
        # 対話履歴を更新する
        self.dialogue_manager.add_message(
            session_id,
            {"role": "user", "content": user_input}
        )
        self.dialogue_manager.add_message(
            session_id,
            {"role": "assistant", "content": response}
        )
       
        return response
   
    def recognize_intent(self, text, history):
        """ユーザーの意図を認識する"""
        prompt = f"""
ユーザーメッセージを分析し、意図のタイプを認識する。

現在のメッセージ：{text}

対話履歴：
{history}

意図タイプ：
- greeting: 挨拶
- inquiry: 問い合わせ
- complaint: 苦情
- order_status: 注文確認
- refund: 返金
- goodbye: 別れの挨拶

意図タイプと信頼度を出力してください。
"""
        result = self.llm.generate(prompt)
        return Intent.parse(result)
   
    def handle_inquiry(self, intent):
        """問い合わせを処理する"""
        # ナレッジベースから関連コンテンツを検索する
        relevant_docs = self.knowledge_base.search(intent.query)
       
        if relevant_docs:
            # 関連ドキュメントがある場合は、ドキュメントに基づいて回答する
            prompt = f"""
以下のドキュメントに基づいてユーザーの質問に回答してください。

ドキュメント：
{relevant_docs}

質問：{intent.query}
"""
            return self.llm.generate(prompt)
        else:
            # 関連ドキュメントがない場合は、オペレーターに転送するか記録する
            return "この問題についてはさらに確認が必要です。少々お待ちください。お調べします。"


class Intent:
    """ユーザーの意図"""
   
    def __init__(self, type, query, confidence):
        self.type = type
        self.query = query
        self.confidence = confidence
   
    @classmethod
    def parse(cls, text):
        """意図テキストを解析する"""
        # 簡略化した実装
        return cls(type="inquiry", query=text, confidence=0.9)

リサーチエージェント

リサーチエージェントは自律的に深い調査を行い、情報を収集・総合できる。

人間のリサーチャーの作業方法を模倣する。

コア能力

検索計画：情報収集の計画を立てる。

多ソース収集：複数のソースから情報を収集する。

情報検証：情報の正確性をクロス検証する。

レポート生成：情報を総合し、構造化されたレポートを生成する。

コード実装

リサーチエージェントの実装

class ResearchAgent:
    """
調査 Agent
自主的に深度調査を行い、レポートを生成する
    """
   
    def __init__(self, search_tools, llm):
        # 検索ツール一覧
        self.search_tools = search_tools
        self.llm = llm
   
    def research(self, topic, depth=3):
        """
調査タスクを実行する
:param topic: 調査テーマ
:param depth: 調査深度（検索ラウンド数）
        """
        # ステップ1：調査計画を立てる
        plan = self.create_research_plan(topic, depth)
       
        # ステップ2：複数ラウンドの検索を実行する
        findings = []
        for phase in plan.phases:
            # 複数の方向を並行して検索する
            phase_results = self.search_phase(phase)
            findings.extend(phase_results)
           
            # 現在の発見を総合する
            synthesis = self.synthesize(findings)
            findings.append({
                "type": "synthesis",
                "content": synthesis,
                "phase": phase.number
            })
       
        # ステップ3：最終レポートを生成する
        report = self.generate_report(findings)
       
        return {
            "topic": topic,
            "plan": plan,
            "findings": findings,
            "report": report
        }
   
    def create_research_plan(self, topic, depth):
        """調査計画を作成"""
        prompt = f"""
以下の調査テーマについて詳細な計画を立てます：

テーマ：{topic}
深度：{depth}

以下を計画してください：
1. 調査が必要な主要な方向性（3〜5つ）
2. 各方向で回答すべき核心的な質問
3. 情報源の提案
"""
        return ResearchPlan.parse(self.llm.generate(prompt))
   
    def search_phase(self, phase):
        """個別の検索段階を実行"""
        results = []
       
        for query in phase.queries:
            # 検索ツールを使用して検索する
            for tool in self.search_tools:
                search_results = tool.search(query)
                results.extend(search_results)
       
        return results
   
    def synthesize(self, findings):
        """発見を総合"""
        prompt = f"""
以下の調査結果を総合し、重要な情報を抽出します：

発見リスト：
{findings}

以下を抽出してください：
1. 主要な結論
2. 結論を裏付ける証拠
3. 存在する論争や不確実性
4. さらなる調査が必要な問題
"""
        return self.llm.generate(prompt)
   
    def generate_report(self, findings):
        """最終レポートを生成"""
        prompt = f"""
以下の調査結果に基づいて、構造化されたレポートを生成します：

調査結果：
{findings}

レポート要件：
1. エグゼクティブサマリー
2. 背景紹介
3. 主要な発見（テーマ別）
4. 結論と提案
5. 参考情報源

完全なレポートを生成してください。
"""
        return self.llm.generate(prompt)


class ResearchPlan:
    """調査計画"""
   
    @classmethod
    def parse(cls, text):
        """計画テキストを解析"""
        # 簡略化された実装
        return cls(phases=[
            ResearchPhase(number=1, queries=["関連テーマ1"], goals=["基本概念を理解する"]),
            ResearchPhase(number=2, queries=["関連テーマ2"], goals=["詳細分析"]),
        ])
   
    def __init__(self, phases):
        self.phases = phases


class ResearchPhase:
    """調査段階"""
   
    def __init__(self, number, queries, goals):
        self.number = number
        self.queries = queries
        self.goals = goals

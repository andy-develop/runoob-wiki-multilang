Función de ordenación de tablas JavaScript/CSS

Categoría

En este capítulo aprenderemos cómo usar JS/CSS para implementarordenación de tablasfunción.

Primero veamos el efecto a continuación, haga clic en el botón para ordenar (orden alfabético ascendente, según el campo Nombre):

Código de ordenación

Ejemplo

function sortTable() { var table, rows, switching, i, x, y, shouldSwitch; table = document.getElementById("myTable"); switching = true; /*Establecer una sentencia de bucle*/ while (switching) { //Establecer el indicador de fin del bucle switching = false; rows = table.rows; /*Recorrer las filas de la tabla*/ for (i = 1; i < (rows.length - 1); i++) { //Establecer si los elementos deben intercambiarse de posición shouldSwitch = false; /*Obtener los elementos a comparar*/ x = rows[i].getElementsByTagName("TD")[0]; y = rows[i + 1].getElementsByTagName("TD")[0]; //Determinar si se debe intercambiar el siguiente elemento con el elemento actual if (x.innerHTML.toLowerCase() > y.innerHTML.toLowerCase()) { //Establecer la marca de intercambio de elementos y finalizar el bucle actual shouldSwitch = true; break; } } if (shouldSwitch) { /*Si el indicador de intercambio de elementos está establecido en true, realizar la operación de intercambio*/ rows[i].parentNode.insertBefore(rows[i + 1], rows[i]); switching = true; } } }

Ordenar haciendo clic en el encabezado de la tabla

Haga clic en el campo del encabezado para ordenar la tabla.

Al hacer clic por primera vez, la dirección de ordenación es ascendente (A a Z).

Al hacer clic de nuevo, la dirección de ordenación es descendente (Z a A).

Ejemplo

function sortTable(n) { var table, rows, switching, i, x, y, shouldSwitch, dir, switchcount = 0; table = document.getElementById("myTable2"); switching = true; //Establecer ordenación ascendente dir = "asc"; /*Establecer una sentencia de bucle*/ while (switching) { //Establecer el indicador de fin del bucle switching = false; rows = table.rows; /*Recorrer la lista, comenzando desde la segunda fila*/ for (i = 1; i < (rows.length - 1); i++) { //Establecer si los elementos deben intercambiarse de posición shouldSwitch = false; /*Obtener los elementos a comparar, uno de la fila actual y uno de la siguiente:*/ x = rows[i].getElementsByTagName("TD")[n]; y = rows[i + 1].getElementsByTagName("TD")[n]; /*Determinar si se debe intercambiar el siguiente elemento con el elemento actual (asc o desc):*/ if (dir == "asc") { if (x.innerHTML.toLowerCase() > y.innerHTML.toLowerCase()) { //Establecer la marca de intercambio de elementos y finalizar el bucle actual shouldSwitch = true; break; } } else if (dir == "desc") { if (x.innerHTML.toLowerCase() < y.innerHTML.toLowerCase()) { //Establecer la marca de intercambio de elementos y finalizar el bucle actual shouldSwitch = true; break; } } } if (shouldSwitch) { /*Si el indicador de intercambio de elementos está establecido en true, realizar la operación de intercambio*/ rows[i].parentNode.insertBefore(rows[i + 1], rows[i]); switching = true; //Cada vez que se complete un intercambio, incremente switchcount en 1 switchcount ++; } else { /*Si se ha completado la ordenación de todos los elementos y direction está establecido en "asc", entonces establezca direction en "desc" y ejecute el bucle nuevamente*/ if (switchcount == 0 && dir == "asc") { dir = "desc"; switching = true; } } } }

Manual de referencia de PyTorch torch.nn

de PyTorchtorch.nnEl módulo es un módulo central para construir y entrenar redes neuronales; proporciona una amplia variedad de clases y funciones para definir y operar redes neuronales.

A continuación se presentantorch.nnalgunos componentes clave del módulo y sus funciones:

1. Clase nn.Module：

  • nn.ModuleEs la clase base de todos los modelos de redes neuronales personalizados. Los usuarios suelen derivar sus propias clases de modelo de esta clase y definir en ella la estructura de las capas de la red y la función de propagación hacia adelante (forward pass).

2. Capas predefinidas (Modules)：

  • Incluye varios tipos de componentes de capas, como capas convolucionales (nn.Conv1d, nn.Conv2d, nn.Conv3d), capas totalmente conectadas (nn.Linear), funciones de activación (nn.ReLU, nn.Sigmoid, nn.Tanh), etc.

3. Clases contenedoras：

  • nn.Sequential: Permiten combinar varias capas en orden para formar redes simples apiladas linealmente.
  • nn.ModuleList 和 nn.ModuleDict: Pueden almacenar y acceder dinámicamente a submódulos, admitiendo colecciones de módulos de longitud variable o con nombre.

4. Funciones de pérdida (Loss Functions)：

  • torch.nnIncluye una serie de funciones de pérdida para medir la diferencia entre las predicciones del modelo y las etiquetas verdaderas, como el error cuadrático medio (nn.MSELoss), la pérdida de entropía cruzada (nn.CrossEntropyLoss), etc.

5. Interfaz funcional (Functional Interface)：

  • nn.functional(Generalmente abreviada comoF), contiene muchas funciones que pueden actuar directamente sobre tensores. Implementan las mismas funcionalidades que los objetos de capa, pero no tienen la capacidad de guardar y actualizar parámetros. Por ejemplo, se puede usarF.relu()para realizar una operación ReLU directamente, oF.conv2d()para realizar una operación de convolución.

6. Métodos de inicialización：

  • torch.nn.initProporcionan algunas estrategias comunes de inicialización de pesos, como la inicialización de Xavier (nn.init.xavier_uniform_()) y la inicialización de Kaiming (nn.init.kaiming_uniform_()), que son cruciales para entrenar redes neuronales con éxito.

7. Capa Transformer：

  • PyTorch proporciona componentes completos de la arquitectura Transformer, incluyendonn.Transformer, nn.TransformerEncoder, nn.TransformerDecodery mecanismos de atenciónnn.MultiheadAttention, etc.

8. Capas de normalización：

  • Incluyen normalización por lotes (BatchNorm), normalización de capa (LayerNorm), normalización de grupo (GroupNorm), normalización de instancia (InstanceNorm) y RMSNorm, etc.

Ejemplo: clasificación de imágenes con CNN

import torch
import torch.nn as nn

class CNN(nn.Module):
    def __init__(self):
        super(CNN, self).__init__()
        # Capa convolucional
        self.conv1 = nn.Conv2d(3, 16, kernel_size=3, padding=1)
        self.conv2 = nn.Conv2d(16, 32, kernel_size=3, padding=1)
        self.conv3 = nn.Conv2d(32, 64, kernel_size=3, padding=1)

        # Capa de pooling
        self.pool = nn.MaxPool2d(2, 2)

        # Normalización por lotes
        self.bn1 = nn.BatchNorm2d(16)
        self.bn2 = nn.BatchNorm2d(32)
        self.bn3 = nn.BatchNorm2d(64)

        # Función de activación
        self.relu = nn.ReLU()

        # Capa totalmente conectada
        self.fc1 = nn.Linear(64 * 4 * 4, 256)
        self.fc2 = nn.Linear(256, 10)

        # Dropout
        self.dropout = nn.Dropout(0.5)

    def forward(self, x):
        # Conv -> BN -> ReLU -> Pool
        x = self.pool(self.relu(self.bn1(self.conv1(x))))
        x = self.pool(self.relu(self.bn2(self.conv2(x))))
        x = self.pool(self.relu(self.bn3(self.conv3(x))))

        # Flatten
        x = x.view(x.size(0), -1)

        # FC -> ReLU -> Dropout -> FC
        x = self.dropout(self.relu(self.fc1(x)))
        x = self.fc2(x)
        return x

# Crear el modelo
model = CNN()
print(model)

# Probar la propagación hacia adelante
input_tensor = torch.randn(1, 3, 32, 32)
output = model(input_tensor)
print("Output shape:", output.shape)

Ejemplo: clasificación de texto con LSTM

import torch
import torch.nn as nn

class LSTMClassifier(nn.Module):
    def __init__(self, vocab_size, embedding_dim, hidden_dim, num_layers, output_dim):
        super(LSTMClassifier, self).__init__()

        # Capa de embedding
        self.embedding = nn.Embedding(vocab_size, embedding_dim, padding_idx=0)

        # Capa LSTM
        self.lstm = nn.LSTM(
            embedding_dim,
            hidden_dim,
            num_layers=num_layers,
            batch_first=True,
            bidirectional=True,
            dropout=0.5
        )

        # Capa totalmente conectada
        self.fc = nn.Linear(hidden_dim * 2, output_dim)

        # Dropout
        self.dropout = nn.Dropout(0.5)

    def forward(self, text, text_lengths):
        # Embedding
        embedded = self.embedding(text)

        # Empaquetar secuencias para manejar entradas de longitud variable
        packed = nn.utils.rnn.pack_padded_sequence(
            embedded, text_lengths.cpu(), batch_first=True, enforce_sorted=False
        )

        # LSTM
        packed_output, (hidden, cell) = self.lstm(packed)

        # Desempaquetar
        output, output_lengths = nn.utils.rnn.pad_packed_sequence(packed_output, batch_first=True)

        # Combinar los últimos estados ocultos de la dirección bidireccional
        hidden = torch.cat((hidden[-2,:,:], hidden[-1,:,:]), dim=1)

        # Dropout y totalmente conectada
        hidden = self.dropout(hidden)
        output = self.fc(hidden)

        return output

# Parámetros
vocab_size = 10000
embedding_dim = 128
hidden_dim = 256
num_layers = 2
output_dim = 5  # 5 clases

# Crear el modelo
model = LSTMClassifier(vocab_size, embedding_dim, hidden_dim, num_layers, output_dim)
print(model)

Ejemplo: uso del codificador Transformer

import torch
import torch.nn as nn

class TransformerClassifier(nn.Module):
    def __init__(self, input_dim, d_model, nhead, num_layers, dim_feedforward, output_dim, dropout):
        super(TransformerClassifier, self).__init__()

        # Capa de embedding
        self.embedding = nn.Linear(input_dim, d_model)

        # Codificación posicional
        self.positional_encoding = nn.Parameter(torch.randn(1, 1000, d_model) * 0.1)

        # Capa del codificador Transformer
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=d_model,
            nhead=nhead,
            dim_feedforward=dim_feedforward,
            dropout=dropout,
            batch_first=True
        )

        # Codificador Transformer
        self.transformer_encoder = nn.TransformerEncoder(encoder_layer, num_layers=num_layers)

        # Cabecera de clasificación
        self.fc = nn.Linear(d_model, output_dim)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        # Agregar codificación posicional
        seq_len = x.size(1)
        x = self.embedding(x) + self.positional_encoding[:, :seq_len, :]

        # Codificación Transformer
        x = self.transformer_encoder(x)

        # Usar la salida de la primera posición para clasificación (similar al token CLS)
        x = x[:, 0, :]

        x = self.dropout(x)
        x = self.fc(x)

        return x

# Parámetros
input_dim = 512
d_model = 512
nhead = 8
num_layers = 6
dim_feedforward = 2048
output_dim = 10
dropout = 0.1

# Crear el modelo
model = TransformerClassifier(input_dim, d_model, nhead, num_layers, dim_feedforward, output_dim, dropout)
print(model)

# Probar
x = torch.randn(32, 100, input_dim)  # batch_size=32, seq_len=100
output = model(x)
print("Output shape:", output.shape)  # (32, 10)

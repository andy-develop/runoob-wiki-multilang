# 导入模块
from PIL import Image, ImageFilter, ImageEnhance, ImageDraw, ImageFont
import os

Ejemplo

El siguiente es un ejemplo completo que muestra las siguientes operaciones:

  1. Abrir y mostrar información básica de la imagen
    • Formato, tamaño, modo, etc.
  2. Ajustar el tamaño de la imagen
    • Ajustar la imagen a las dimensiones especificadas
  3. Recortar imagen
    • Extraer un área específica de la imagen original
  4. Rotar imagen
    • Rotar 45 grados y conservar la imagen completa
  5. Voltear imagen
    • Voltear horizontal y verticalmente
  6. Convertir modo de color
    • Convertir la imagen en color a escala de grises
  7. Aplicar filtros
    • Efectos como desenfoque, realce de bordes, contorno, etc.
  8. Ajustar propiedades de la imagen
    • Ajustes de brillo, contraste y nitidez
  9. Añadir texto
    • Dibujar texto sobre la imagen
  10. Combinar imágenes
    • Unión simple de imágenes
  11. Guardar en diferentes formatos y calidades
    • PNG, BMP y JPEG de diferentes calidades

Ejemplo

"""
Tutorial de operaciones básicas de imagen con Python Pillow
Imagen de prueba utilizada: tiger.jpeg
"""

from PIL import Image, ImageFilter, ImageEnhance, ImageDraw, ImageFont
import os

def main():
    # Asegurarse de que haya una carpeta de salida en nuestro directorio de trabajo
    output_dir = "output"
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
   
    # 1. Abrir imagen
    print("1. Abrir imagen")
    try:
        img = Image.open("tiger.jpeg")
        print(f"Imagen abierta correctamente - Formato: {img.format}, Tamaño: {img.size}, Modo: {img.mode}")
    except Exception as e:
        print(f"No se pudo abrir la imagen: {e}")
        return
   
    # 2. Mostrar información básica de la imagen
    print("\n2. Información básica de la imagen")
    print(f"Formato de imagen: {img.format}")
    print(f"Tamaño de imagen (ancho x alto): {img.size}")
    print(f"Modo de imagen: {img.mode}")
   
    # 3. Ajustar el tamaño de la imagen
    print("\n3. Ajustar el tamaño de la imagen")
    resized_img = img.resize((400, 300))
    resized_img.save(f"{output_dir}/resized_tiger.jpg")
    print("Se ha guardado la imagen ajustada")
   
    # 4. Recortar imagen
    print("\n4. Recortar imagen")
    # Área de recorte (izquierda, superior, derecha, inferior)
    width, height = img.size
    crop_area = (width//4, height//4, 3*width//4, 3*height//4)  # Recortar el área central
    cropped_img = img.crop(crop_area)
    cropped_img.save(f"{output_dir}/cropped_tiger.jpg")
    print(f"Se ha recortado el área {crop_area} y se ha guardado")
   
    # 5. Rotar imagen
    print("\n5. Rotar imagen")
    rotated_img = img.rotate(45, expand=True)  # Rotar 45 grados, expand=True conserva toda la imagen
    rotated_img.save(f"{output_dir}/rotated_tiger.jpg")
    print("Se ha guardado la imagen rotada")
   
    # 6. Voltear imagen
    print("\n6. Voltear imagen")
    # Voltear horizontalmente
    h_flipped = img.transpose(Image.FLIP_LEFT_RIGHT)
    h_flipped.save(f"{output_dir}/horizontal_flip_tiger.jpg")
    # Voltear verticalmente
    v_flipped = img.transpose(Image.FLIP_TOP_BOTTOM)
    v_flipped.save(f"{output_dir}/vertical_flip_tiger.jpg")
    print("Se han guardado las imágenes volteadas horizontal y verticalmente")
   
    # 7. Convertir modo de color
    print("\n7. Convertir modo de color")
    grayscale_img = img.convert('L')  # Convertir a imagen en escala de grises
    grayscale_img.save(f"{output_dir}/grayscale_tiger.jpg")
    print("Se ha guardado la imagen en escala de grises")
   
    # 8. Aplicar filtros
    print("\n8. Aplicar filtros")
    # Filtro de desenfoque
    blurred_img = img.filter(ImageFilter.BLUR)
    blurred_img.save(f"{output_dir}/blurred_tiger.jpg")
    # Realce de bordes
    edge_img = img.filter(ImageFilter.EDGE_ENHANCE)
    edge_img.save(f"{output_dir}/edge_enhanced_tiger.jpg")
    # Filtro de contorno
    contour_img = img.filter(ImageFilter.CONTOUR)
    contour_img.save(f"{output_dir}/contour_tiger.jpg")
    print("Se han guardado las imágenes con diferentes filtros aplicados")
   
    # 9. Ajustar brillo, contraste y nitidez
    print("\n9. Ajustar propiedades de la imagen")
    # Aumentar brillo
    brightness = ImageEnhance.Brightness(img)
    bright_img = brightness.enhance(1.5)  # Aumentar el brillo en un 50%
    bright_img.save(f"{output_dir}/brighter_tiger.jpg")
   
    # Aumentar contraste
    contrast = ImageEnhance.Contrast(img)
    contrast_img = contrast.enhance(1.5)  # Aumentar el contraste en un 50%
    contrast_img.save(f"{output_dir}/contrast_tiger.jpg")
   
    # Aumentar nitidez
    sharpness = ImageEnhance.Sharpness(img)
    sharp_img = sharpness.enhance(2.0)  # Aumentar la nitidez en un 100%
    sharp_img.save(f"{output_dir}/sharper_tiger.jpg")
    print("Se han guardado las imágenes con brillo, contraste y nitidez ajustados")
   
    # 10. Añadir texto
    print("\n10. Añadir texto")
    # Crear una copia para dibujar
    text_img = img.copy()
    draw = ImageDraw.Draw(text_img)
    # Intentar usar la fuente del sistema; si falla, usar la predeterminada
    try:
        # Para diferentes sistemas operativos, pueden ser necesarias diferentes rutas de fuente
        # Ejemplo de ruta de fuente en Windows
        font = ImageFont.truetype("arial.ttf", 36)
    except IOError:
        font = ImageFont.load_default()
   
    # Añadir texto sobre la imagen
    draw.text((10, 10), "Tiger", fill=(255, 255, 255), font=font)
    text_img.save(f"{output_dir}/text_tiger.jpg")
    print("Se ha guardado la imagen con texto añadido")
   
    # 11. Combinar imágenes (unión simple)
    print("\n11. Combinar imágenes")
    # Crear una nueva imagen, con el doble de ancho de la imagen original y la misma altura
    merged_img = Image.new('RGB', (width*2, height))
    # Colocar la imagen original a la izquierda
    merged_img.paste(img, (0, 0))
    # Colocar la imagen en escala de grises a la derecha
    merged_img.paste(grayscale_img.convert('RGB'), (width, 0))
    merged_img.save(f"{output_dir}/merged_tiger.jpg")
    print("Se ha guardado la imagen combinada")
   
    # 12. Guardar en diferentes formatos
    print("\n12. Guardar en diferentes formatos")
    img.save(f"{output_dir}/tiger.png")  # Formato PNG
    img.save(f"{output_dir}/tiger.bmp")  # Formato BMP
   
    # Guardar como JPEG, configurar la calidad
    img.save(f"{output_dir}/tiger_high_quality.jpg", quality=95)  # Alta calidad
    img.save(f"{output_dir}/tiger_low_quality.jpg", quality=10)   # Baja calidad
    print("Se han guardado imágenes en diferentes formatos y calidades")
   
    print("\n¡Todas las operaciones completadas! Las imágenes procesadas se guardan en la carpeta 'output'.")

if __name__ == "__main__":
    main()

El código adopta un enfoque estructurado, cada operación tiene comentarios claros. Todas las imágenes procesadas se guardarán enoutputel directorio, para facilitar la visualización y comparación de efectos.

Método de uso:

  • Asegúrate de tener la imagen tiger.jpeg en el directorio donde se encuentra el código
  • Ejecuta el código y se generarán varias imágenes procesadas en la carpeta 'output'
  • Observa los diferentes efectos según la información de salida

Instrucciones de operación

1. Abrir imagen y obtener información básica

Ejemplo

# Abrir archivo de imagen
img = Image.open("tiger.jpeg")

# Obtener y mostrar información de la imagen
print(f"Formato de imagen: {img.format}")  # Por ejemplo: JPEG, PNG, etc.
print(f"Tamaño de imagen (ancho x alto): {img.size}")  # Por ejemplo: (800, 600)
print(f"Modo de imagen: {img.mode}")  # Por ejemplo: RGB, RGBA, L (escala de grises), etc.

Explicación detallada：

  • Image.open()La función se utiliza para abrir el archivo de imagen; el parámetro es la ruta del archivo de imagen.
  • img.formatMuestra el formato de la imagen (como JPEG, PNG, GIF, etc.)
  • img.sizeDevuelve una tupla que contiene el ancho y la altura, como (800, 600)
  • img.modeMuestra el modo de color de la imagen:
    • RGB: Imagen en color
    • RGBA: Imagen en color con canal alfa
    • L: Imagen en escala de grises
    • 1: Imagen binaria (blanco y negro)

2. Ajustar el tamaño de la imagen

Ejemplo

# Ajustar el tamaño de la imagen a 400x300 píxeles
resized_img = img.resize((400, 300))

# Guardar la imagen ajustada
resized_img.save("output/resized_tiger.jpg")

Explicación detallada：

  • resize()El método se utiliza para cambiar las dimensiones de la imagen; el parámetro es una tupla que contiene el nuevo ancho y alto.
  • Este método no modifica la imagen original, sino que devuelve un nuevo objeto de imagen.
  • Por defecto utiliza el método de interpolación de vecino más cercano; para un ajuste de mayor calidad se puede añadir el parámetro:resized_img = img.resize((400, 300), Image.LANCZOS)
  • save()El método guarda la imagen procesada en la ruta especificada.

3. Recortar imagen

Ejemplo

# Obtener dimensiones de la imagen original
width, height = img.size

# Definir el área de recorte (izquierda, superior, derecha, inferior)
crop_area = (width//4, height//4, 3*width//4, 3*height//4)  # Recortar el área central

# Ejecutar el recorte
cropped_img = img.crop(crop_area)

# Guardar la imagen recortada
cropped_img.save("output/cropped_tiger.jpg")

Explicación detallada：

  • crop()El método se utiliza para extraer un área rectangular de la imagen.
  • El área de recorte está compuesta por la tupla(left, upper, right, lower)Definición:
    • left: Coordenada x del límite izquierdo
    • upper: Coordenada y del límite superior
    • right: Coordenada x del límite derecho
    • lower: Coordenada y del límite inferior
  • En el ejemplo, recortamos la parte central de la imagen (área de 1/4 a 3/4)
  • width//4de//es la división entera, que asegura que el resultado sea un número entero

4. Rotar imagen

Ejemplo

# Rotar la imagen 45 grados y expandir el lienzo para mostrar la imagen completa
rotated_img = img.rotate(45, expand=True)

# Guardar la imagen rotada
rotated_img.save("output/rotated_tiger.jpg")

Explicación detallada：

  • rotate()El método se usa para rotar la imagen; el primer parámetro es el ángulo de rotación (en sentido antihorario, en grados)
  • expand=TrueEl parámetro expande las dimensiones de la imagen de salida para acomodar toda la imagen rotada
  • Si no se estableceexpand=True, la imagen rotada se recortará para ajustarse al tamaño original
  • Por defecto, el centro de rotación es el punto central de la imagen

5. Voltear imagen

Ejemplo

# Volteo horizontal (espejo izquierda-derecha)
h_flipped = img.transpose(Image.FLIP_LEFT_RIGHT)
h_flipped.save("output/horizontal_flip_tiger.jpg")

# Volteo vertical (espejo arriba-abajo)
v_flipped = img.transpose(Image.FLIP_TOP_BOTTOM)
v_flipped.save("output/vertical_flip_tiger.jpg")

Explicación detallada：

  • transpose()El método se usa para operaciones de volteo y transposición de imágenes
  • Image.FLIP_LEFT_RIGHTLa constante representa el volteo horizontal (espejo izquierda-derecha)
  • Image.FLIP_TOP_BOTTOMLa constante representa el volteo vertical (espejo arriba-abajo)
  • Además del volteo, se pueden usar otras constantes comoImage.ROTATE_90, Image.ROTATE_180, Image.ROTATE_270para rotar

6. Convertir modo de color

Ejemplo

# Convertir la imagen a escala de grises
grayscale_img = img.convert('L')

# Guardar la imagen en escala de grises
grayscale_img.save("output/grayscale_tiger.jpg")

Explicación detallada：

  • convert()El método se usa para cambiar el modo de la imagen (espacio de color)
  • El parámetro'L'representa la conversión a imagen en escala de grises (Luminance)
  • Otros modos comunes incluyen:
    • 'RGB': Color de tres canales
    • 'RGBA': Color de cuatro canales (con transparencia)
    • 'CMYK': Modo de cuatro colores de impresión
    • '1': Imagen binaria (blanco y negro puro)
  • Cambiar el modo puede reducir el tamaño del archivo o preparar la imagen para un procesamiento específico

7. Aplicar filtros de imagen

Ejemplo

# Aplicar filtro de desenfoque
blurred_img = img.filter(ImageFilter.BLUR)
blurred_img.save("output/blurred_tiger.jpg")

# Aplicar filtro de realce de bordes
edge_img = img.filter(ImageFilter.EDGE_ENHANCE)
edge_img.save("output/edge_enhanced_tiger.jpg")

# Aplicar filtro de contorno
contour_img = img.filter(ImageFilter.CONTOUR)
contour_img.save("output/contour_tiger.jpg")

Explicación detallada：

  • filter()El método se usa para aplicar varios efectos de filtro predefinidos a la imagen
  • ImageFilter.BLURDesenfoca la imagen, similar a un efecto de enfoque suave
  • ImageFilter.EDGE_ENHANCERealza los bordes de la imagen, haciendo que los contornos sean más visibles
  • ImageFilter.CONTOUREncuentra las líneas de contorno de la imagen, produciendo un efecto similar a un boceto
  • Otros filtros comunes incluyen:
    • ImageFilter.SHARPEN: Afinar la imagen
    • ImageFilter.EMBOSS: Efecto de relieve
    • ImageFilter.FIND_EDGES: Detección de bordes
    • ImageFilter.SMOOTH: Suavizar la imagen
  • También se pueden crear filtros personalizados, como el desenfoque gaussiano:img.filter(ImageFilter.GaussianBlur(radius=2))

8. Ajustar propiedades de la imagen (brillo, contraste, nitidez)

Ejemplo

# Aumentar el brillo (1.5 significa que el brillo aumenta un 50%)
brightness = ImageEnhance.Brightness(img)
bright_img = brightness.enhance(1.5)
bright_img.save("output/brighter_tiger.jpg")

# Aumentar el contraste (1.5 significa que el contraste aumenta un 50%)
contrast = ImageEnhance.Contrast(img)
contrast_img = contrast.enhance(1.5)
contrast_img.save("output/contrast_tiger.jpg")

# Aumentar la nitidez (2.0 significa que la nitidez aumenta un 100%)
sharpness = ImageEnhance.Sharpness(img)
sharp_img = sharpness.enhance(2.0)
sharp_img.save("output/sharper_tiger.jpg")

Explicación detallada：

  • ImageEnhanceEl módulo proporciona varias clases de mejoradores para ajustar diferentes propiedades de la imagen
  • Cada mejorador primero debe inicializarse con un objeto de imagen
  • enhance()El método acepta un parámetro que representa el grado de mejora:
    • Un valor de 1.0 significa que la imagen original no cambia
    • Un valor mayor que 1.0 indica un efecto de mejora (por ejemplo, 1.5 significa un aumento del 50%)
    • Un valor menor que 1.0 indica un efecto de reducción (por ejemplo, 0.5 significa una disminución del 50%)
  • Además de los tres mejoradores mostrados arriba, también estáImageEnhance.Colorpara ajustar la saturación de color

9. Añadir texto a la imagen

Ejemplo

# Crear una copia de la imagen para dibujar
text_img = img.copy()

# Crear el objeto de dibujo
draw = ImageDraw.Draw(text_img)

# Intentar cargar la fuente (si no está disponible, usar la fuente predeterminada)
try:
    # Cargar una fuente del sistema, tamaño 36 píxeles
    font = ImageFont.truetype("arial.ttf", 36)
except IOError:
    # Si no se encuentra la fuente especificada, usar la fuente predeterminada
    font = ImageFont.load_default()

# Añadir texto a la imagen
draw.text((10, 10), "Tiger", fill=(255, 255, 255), font=font)

# Guardar la imagen con el texto añadido
text_img.save("output/text_tiger.jpg")

Explicación detallada：

  • copy()El método crea una copia de la imagen original para evitar modificar la imagen original
  • ImageDraw.Draw()Crea un objeto de dibujo para dibujar sobre la imagen
  • ImageFont.truetype()Se usa para cargar fuentes TrueType:
    • El primer parámetro es la ruta del archivo de fuente (por ejemplo, "arial.ttf")
    • El segundo parámetro es el tamaño de la fuente (en píxeles)
  • draw.text()Se usa para dibujar texto en la imagen:
    • El primer parámetro(10, 10)son las coordenadas de posición del texto (esquina superior izquierda)
    • El segundo parámetro es el contenido de texto a dibujar
    • fillEl parámetro especifica el color del texto; aquí es blanco(255, 255, 255)
    • fontEl parámetro especifica la fuente a utilizar
  • Usar el bloquetry-exceptpara manejar posibles fallos al cargar la fuente

10. Combinar imágenes (unión simple)

Ejemplo

# Obtener las dimensiones de la imagen original
width, height = img.size

# Crear una nueva imagen en blanco con el doble de ancho que la imagen original
merged_img = Image.new('RGB', (width*2, height))

# Pegar la imagen original en el lado izquierdo
merged_img.paste(img, (0, 0))

# Convertir la imagen en escala de grises a RGB y pegarla en el lado derecho
merged_img.paste(grayscale_img.convert('RGB'), (width, 0))

# Guardar la imagen combinada
merged_img.save("output/merged_tiger.jpg")

Explicación detallada：

  • Image.new()Crear una nueva imagen en blanco:
    • El primer parámetro es el modo de color; aquí es'RGB'
    • El segundo parámetro son las dimensiones de la imagen; aquí es el doble del ancho de la imagen original y la altura de la imagen original
  • paste()El método pega una imagen sobre otra:
    • El primer parámetro es la imagen a pegar
    • El segundo parámetro son las coordenadas de la posición de pegado (esquina superior izquierda)
  • convert('RGB')Convertir la imagen en escala de grises a modo RGB, porque solo se pueden pegar imágenes del mismo modo juntas
  • Este método se puede usar para crear collages, comparaciones lado a lado o crear álbumes de fotos

11. Guardar en diferentes formatos y calidades

Ejemplo

# Guardar en formato PNG (compresión sin pérdida)
img.save("output/tiger.png")

# Guardar en formato BMP (sin compresión)
img.save("output/tiger.bmp")

# Guardar como JPEG de alta calidad (calidad 95%)
img.save("output/tiger_high_quality.jpg", quality=95)

# Guardar como JPEG de baja calidad (calidad 10%)
img.save("output/tiger_low_quality.jpg", quality=10)

Explicación detallada：

  • save()El método determina automáticamente el formato de guardado según la extensión del archivo
  • Características de los diferentes formatos:
    • PNG: Compresión sin pérdida, soporta transparencia, adecuado para iconos, capturas de pantalla, etc.
    • JPEG: Compresión con pérdida, adecuado para fotografías, no soporta transparencia
    • BMP: Sin compresión, archivos más grandes, pero conserva toda la información
    • GIF: Soporta animaciones, hasta 256 colores
  • Para el formato JPEG, se puede controlar la calidad de compresión mediante elqualityparámetro:
    • El valor varía de 1 (peor calidad, archivo más pequeño) a 95 (mejor calidad, archivo más grande)
    • El valor predeterminado suele ser 75
  • Diferentes configuraciones de calidad pueden ayudar a equilibrar la calidad de la imagen y el tamaño del archivo

12. Crear directorio de salida

Ejemplo

# Asegurar que el directorio de salida exista
output_dir = "output"
if not os.path.exists(output_dir):
    os.makedirs(output_dir)

Explicación detallada：

  • os.path.exists()Comprueba si el directorio especificado existe
  • os.makedirs()Crea el directorio si no existe
  • Este código asegura que tengamos un lugar para guardar las imágenes procesadas
  • Colocar todas las imágenes generadas en un directorio separado permite organizar mejor el proyecto

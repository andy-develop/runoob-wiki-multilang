Filtros y mejora de imágenes

Aplicación de filtros integrados (filter())

Pillow proporciona varios filtros integrados, usando el módulo ImageFilter:

Ejemplo

from PIL import Image, ImageFilter

# Abrir imagen
image = Image.open("input.jpg")

# Aplicar filtro de desenfoque
blurred = image.filter(ImageFilter.BLUR)
blurred.save("blurred.jpg")

# Aplicar filtro de contorno
contour = image.filter(ImageFilter.CONTOUR)
contour.save("contour.jpg")

# Aplicar filtro de relieve
emboss = image.filter(ImageFilter.EMBOSS)
emboss.save("emboss.jpg")

# Encontrar bordes
edges = image.filter(ImageFilter.FIND_EDGES)
edges.save("edges.jpg")

# Filtro de nitidez
sharpen = image.filter(ImageFilter.SHARPEN)
sharpen.save("sharpen.jpg")

# Filtro de suavizado
smooth = image.filter(ImageFilter.SMOOTH)
smooth.save("smooth.jpg")

# Mejora de detalles
detail = image.filter(ImageFilter.DETAIL)
detail.save("detail.jpg")

Mejora de imágenes (módulo ImageEnhance)

El módulo ImageEnhance proporciona herramientas para controlar brillo, contraste, color y nitidez:

Ejemplo

from PIL import Image, ImageEnhance

# Abrir imagen
image = Image.open("input.jpg")

# Aumentar brillo (factor > 1 aumenta el brillo, < 1 lo reduce)
enhancer = ImageEnhance.Brightness(image)
brightened = enhancer.enhance(1.5)  # Aumentar el brillo en un 50%
brightened.save("brightened.jpg")

# Aumentar contraste
enhancer = ImageEnhance.Contrast(image)
contrast = enhancer.enhance(1.8)  # Aumentar el contraste en un 80%
contrast.save("contrast.jpg")

# Aumentar la saturación de color
enhancer = ImageEnhance.Color(image)
saturated = enhancer.enhance(1.5)  # Aumentar la saturación en un 50%
saturated.save("saturated.jpg")

# Aumentar nitidez
enhancer = ImageEnhance.Sharpness(image)
sharpened = enhancer.enhance(2.0)  # Aumentar la nitidez al 100%
sharpened.save("sharpened.jpg")

Detección de bordes y nitidez

Además de usar filtros integrados, también se pueden implementar efectos más avanzados de detección de bordes y nitidez:

Ejemplo

from PIL import Image, ImageFilter, ImageChops, ImageOps

# Abrir imagen
image = Image.open("input.jpg")

# Detección de bordes Sobel
# Filtro Sobel horizontal
h_sobel = ImageFilter.Kernel(
    size=(3, 3),
    kernel=[-1, 0, 1, -2, 0, 2, -1, 0, 1],
    scale=1
)
# Filtro Sobel vertical
v_sobel = ImageFilter.Kernel(
    size=(3, 3),
    kernel=[-1, -2, -1, 0, 0, 0, 1, 2, 1],
    scale=1
)

h_edges = image.filter(h_sobel)
v_edges = image.filter(v_sobel)

# Combinar bordes horizontal y vertical
edges = ImageChops.add(
    ImageChops.multiply(h_edges, h_edges),
    ImageChops.multiply(v_edges, v_edges)
)
edges.save("sobel_edges.jpg")

# Usar realce de alta frecuencia para nitidez (USM - Unsharp Masking)
def unsharp_mask(image, radius=2, percent=150, threshold=3):
    """Aplicar filtro de nitidez USM
   
Parámetros:
radius: radio del desenfoque gaussiano
percent: porcentaje de intensidad de nitidez
threshold: umbral mínimo de cambio de brillo para aplicar nitidez
    """
    blurred = image.filter(ImageFilter.GaussianBlur(radius=radius))
    sharpened = Image.blend(image, ImageChops.subtract(image, blurred), percent/100)
    return sharpened

usm_image = unsharp_mask(image, radius=2, percent=200, threshold=5)
usm_image.save("usm_sharpened.jpg")

Dibujo de formas complejas

Ejemplo

from PIL import Image, ImageDraw
import math

# Crear lienzo en blanco
width, height = 800, 800
image = Image.new("RGB", (width, height), color="white")
draw = ImageDraw.Draw(image)

# Dibujar estrella
def draw_star(draw, center, points=5, outer_radius=100, inner_radius=50, rotation=0, **kwargs):
    """Dibujar estrella
   
Parámetros:
center: tupla de coordenadas del punto central (x, y)
points: número de puntas de la estrella
outer_radius: radio del círculo exterior
inner_radius: radio del círculo interior
rotation: ángulo de rotación (grados)
**kwargs: parámetros pasados a polygon()
    """
    cx, cy = center
    angle = math.pi / points
    rotation_rad = math.radians(rotation)
   
    vertices = []
    for i in range(2 * points):
        radius = outer_radius if i % 2 == 0 else inner_radius
        theta = i * angle + rotation_rad
        x = cx + radius * math.sin(theta)
        y = cy - radius * math.cos(theta)
        vertices.append((x, y))
   
    draw.polygon(vertices, **kwargs)

# Dibujar la estrella
draw_star(draw, center=(200, 200), fill="gold", outline="orange", width=2)
draw_star(draw, center=(500, 200), points=8, rotation=22.5,
          outer_radius=120, inner_radius=40, fill="blue", outline="navy", width=2)

# Dibujar corazón
def draw_heart(draw, center, size=100, **kwargs):
    """Dibujar corazón
   
Parámetros:
center: tupla de coordenadas del punto central (x, y)
size: tamaño del corazón
**kwargs: parámetros pasados a polygon()
    """
    cx, cy = center
    vertices = []
    for t in range(100):
        angle = t / 100 * 2 * math.pi
        x = 16 * math.sin(angle) ** 3
        y = 13 * math.cos(angle) - 5 * math.cos(2*angle) - 2 * math.cos(3*angle) - math.cos(4*angle)
        # Escalar y desplazar
        vertices.append((cx + x * size / 16, cy - y * size / 16))
   
    draw.polygon(vertices, **kwargs)

draw_heart(draw, center=(200, 500), size=150, fill="red", outline="darkred", width=3)

# Dibujar espiral
def draw_spiral(draw, center, loops=3, radius_start=5, radius_end=100, points=500, **kwargs):
    """Dibujar espiral
   
Parámetros:
center: tupla de coordenadas del punto central (x, y)
loops: número de vueltas de la espiral
radius_start: radio inicial
radius_end: radio final
points: número de puntos
**kwargs: parámetros pasados a line()
    """
    cx, cy = center
    vertices = []
   
    for i in range(points + 1):
        # Calcular ángulo y radio actuales
        angle = i / points * loops * 2 * math.pi
        radius = radius_start + (radius_end - radius_start) * i / points
       
        x = cx + radius * math.cos(angle)
        y = cy + radius * math.sin(angle)
        vertices.append((x, y))
   
    # Dibujar polilínea
    for i in range(len(vertices) - 1):
        draw.line([vertices[i], vertices[i+1]], **kwargs)

draw_spiral(draw, center=(500, 500), loops=5, fill="purple", width=2)

image.save("complex_shapes.png")

# Crear efecto de texto artístico
def draw_gradient_text(draw, text, position, font, start_color, end_color, steps=10):
    """Dibujar texto con degradado
   
Parámetros:
draw: objeto ImageDraw
text: texto a dibujar
position: posición del texto (x, y)
font: objeto de fuente
start_color: color inicial (r, g, b)
end_color: color final (r, g, b)
steps: número de pasos del degradado
    """
    x, y = position
    width, height = draw.textsize(text, font=font)
   
    # Calcular el ancho de cada carácter
    char_width = width / len(text)
   
    for i, char in enumerate(text):
        # Calcular el color del carácter actual
        progress = i / (len(text) - 1) if len(text) > 1 else 0
        r = int(start_color[0] + (end_color[0] - start_color[0]) * progress)
        g = int(start_color[1] + (end_color[1] - start_color[1]) * progress)
        b = int(start_color[2] + (end_color[2] - start_color[2]) * progress)
        color = (r, g, b)
       
        # Dibujar el carácter actual
        draw.text((x + i * char_width, y), char, font=font, fill=color)

from PIL import Image

# Crear una imagen con canal alfa
width, height = 500, 500
image = Image.new("RGBA", (width, height), color=(0, 0, 0, 0))  # Completamente transparente

# Generar un efecto de transparencia degradada
for y in range(height):
    for x in range(width):
        # La transparencia disminuye de izquierda a derecha (de totalmente opaco a totalmente transparente)
        alpha = 255 - int(255 * x / width)
        # El color degrada de arriba hacia abajo
        r = int(255 * y / height)
        g = int(255 * (1 - y / height))
        b = 128
        image.putpixel((x, y), (r, g, b, alpha))

image.save("alpha_gradient.png")

# Convertir una imagen RGB en una imagen con canal alfa
def add_alpha_channel(image, alpha_value=128):
    """Añadir un canal alfa a una imagen RGB
   
Parámetros:
image: objeto Image de PIL
alpha_value: valor de transparencia (0-255)
    """
    if image.mode != 'RGBA':
        # Convertir al modo RGBA
        rgba_image = image.convert("RGBA")
       
        # Obtener los datos de píxeles
        data = rgba_image.getdata()
       
        # Establecer la transparencia
        new_data = []
        for item in data:
            # Modificar el valor alfa de los píxeles
            if len(item) == 4:  # Ya es RGBA
                new_data.append((item[0], item[1], item[2], alpha_value))
            else:  # RGB
                new_data.append((item[0], item[1], item[2], alpha_value))
       
        rgba_image.putdata(new_data)
        return rgba_image
    return image

# Crear una imagen RGB de ejemplo
rgb_image = Image.new("RGB", (300, 300), color="blue")
# Añadir el canal alfa
rgba_image = add_alpha_channel(rgb_image, alpha_value=128)
rgba_image.save("with_alpha.png")

# Crear una máscara circular
def create_circular_mask(image, radius=None):
    """Crear una máscara circular
   
Parámetros:
image: objeto Image de PIL
radius: radio del círculo, por defecto la mitad del lado más corto de la imagen
    """
    width, height = image.width, image.height
    center_x, center_y = width // 2, height // 2
   
    if radius is None:
        radius = min(center_x, center_y)
   
    # Crear una nueva imagen transparente
    mask = Image.new('L', (width, height), 0)
   
    # Crear la máscara circular
    for y in range(height):
        for x in range(width):
            # Calcular la distancia al centro
            distance = ((x - center_x) ** 2 + (y - center_y) ** 2) ** 0.5
           
            # Si está dentro del círculo, establecer como opaco
            if distance <= radius:
                mask.putpixel((x, y), 255)
   
    # Aplicar la máscara a la imagen original
    if image.mode != 'RGBA':
        image = image.convert('RGBA')
   
    result = Image.new('RGBA', (width, height), (0, 0, 0, 0))
    result.paste(image, (0, 0), mask)
   
    return result

# Crear una imagen de ejemplo
square_image = Image.new("RGB", (300, 300), color="green")
# Aplicar la máscara circular
circular_image = create_circular_mask(square_image)
circular_image.save("circular_image.png")

def custom_blend(image1, image2, mode="soft_light"):
    """Modo de fusión personalizado
   
Parámetros:
image1, image2: las dos imágenes a fusionar
mode: nombre del modo de fusión
    """
    if image1.mode != image2.mode:

    result = Image.new(image1.mode, (width, height))

        for x in range(width):

                g = (g1 * (255 - g2) + g2 * g2 // 255) // 255
                b = (b1 * (255 - b2) + b2 * b2 // 255) // 255

                else:

                else:

            elif mode == "overlay":
                # Implementar el modo de fusión de superposición
                if r1 < 128:
                    r = (2 * r1 * r2) // 255

                r = (r1 + r2) // 2

                b = (b1 + b2) // 2

def transition_blend(image1, image2, direction="horizontal", steps=10):
    """Crear un efecto de transición suave entre dos imágenes
   
Parámetros:
image1, image2: las dos imágenes a fusionar
direction: dirección de la transición, "horizontal" o "vertical"
steps: número de pasos de la transición
    """
    if image1.size != image2.size:
        raise ValueError("Las dos imágenes deben tener el mismo tamaño")
   
    width, height = image1.size
    result = Image.new(image1.mode, (width, height))
   
    for y in range(height):
        for x in range(width):
            if direction == "horizontal":
                # Transición horizontal: de izquierda a derecha
                blend_factor = x / width
            elif direction == "vertical":
                # Transición vertical: de arriba hacia abajo
                blend_factor = y / height
            elif direction == "radial":

                center_x, center_y = width // 2, height // 2

                max_distance = max((width**2 + height**2)**0.5 / 2, 1)
                blend_factor = min(1.0, distance / max_distance)
            else:
                blend_factor = 0.5  # Por defecto, fusión uniforme
           
            # Obtener los píxeles de ambas imágenes
            r1, g1, b1 = image1.getpixel((x, y))
            r2, g2, b2 = image2.getpixel((x, y))

            g = int(g1 * (1 - blend_factor) + g2 * blend_factor)
            b = int(b1 * (1 - blend_factor) + b2 * blend_factor)
           
            result.putpixel((x, y), (r, g, b))
   
    return result

# Crear un efecto de fusión fluida
def fluid_blend(image1, image2, turbulence=10.0, seed=42):
    """Crear un efecto de fusión de tipo fluido
   
Parámetros:
image1, image2: las dos imágenes a fusionar
turbulence: intensidad de la turbulencia
seed: semilla aleatoria
    """
    if image1.size != image2.size:
        raise ValueError("Las dos imágenes deben tener el mismo tamaño")
   
    import random
    random.seed(seed)
    width, height = image1.size
    result = Image.new(image1.mode, (width, height))
   
    # Crear una implementación simplificada del ruido de Perlin
    def noise(x, y):
        # Función de ruido simplificada
        n = x + y * 57
        n = (n << 13) ^ n
        return (1.0 - ((n * (n * n * 15731 + 789221) + 1376312589) & 0x7fffffff) / 1073741824.0)
   
    for y in range(height):
        for x in range(width):
            # Usar coordenadas distorsionadas para generar el factor de fusión
            noise_value = noise(x / turbulence, y / turbulence)
            blend_factor = (noise_value + 1.0) / 2.0  # Normalizar al rango 0-1
           
            # Obtener los píxeles de ambas imágenes
            r1, g1, b1 = image1.getpixel((x, y))
            r2, g2, b2 = image2.getpixel((x, y))
           
            # Calcular el píxel fusionado
            r = int(r1 * (1 - blend_factor) + r2 * blend_factor)
            g = int(g1 * (1 - blend_factor) + g2 * blend_factor)
            b = int(b1 * (1 - blend_factor) + b2 * blend_factor)
           
            result.putpixel((x, y), (r, g, b))
   
    return result

def create_gradient_mask(size, direction="horizontal", start=0, end=255):
    """Crear una máscara de degradado
   
Parámetros:
size: dimensiones de la máscara (width, height)
direction: dirección del degradado "horizontal", "vertical", "radial", "diagonal"
start: valor de brillo inicial (0-255)
end: valor de brillo final (0-255)
    """
    width, height = size
    mask = Image.new('L', size, 0)
   
    for y in range(height):
        for x in range(width):
            if direction == "horizontal":
                # Degradado horizontal
                value = start + int((end - start) * x / width)
            elif direction == "vertical":
                # Degradado vertical
                value = start + int((end - start) * y / height)
            elif direction == "radial":
                # Degradado radial
                center_x, center_y = width // 2, height // 2
                max_dist = math.sqrt((width/2)**2 + (height/2)**2)
                dist = math.sqrt((x - center_x)**2 + (y - center_y)**2)
                value = start + int((end - start) * (dist / max_dist))
            elif direction == "diagonal":
                # Degradado diagonal
                value = start + int((end - start) * (x + y) / (width + height))
            else:
                value = 0
           
            # Asegurar que los valores estén en el rango 0-255
            value = max(0, min(255, value))
            mask.putpixel((x, y), value)
   
    return mask

def apply_mask_blend(image1, image2, mask):

                b = int(b1 * (1 - blend_factor) + b2 * blend_factor)
               
                result.putpixel((x, y), (r, g, b))
            elif len(pixel1) == 4:  # RGBA

                r2, g2, b2, a2 = pixel2
               
                r = int(r1 * (1 - blend_factor) + r2 * blend_factor)

                b = int(b1 * (1 - blend_factor) + b2 * blend_factor)

# Crear una máscara de patrón
def create_pattern_mask(size, pattern_type="grid", scale=40):
    """Crear una máscara de patrón
   
Parámetros:
size: dimensiones de la máscara (width, height)
pattern_type: tipo de patrón "grid", "dots", "stripes", "waves"
scale: factor de escala del patrón
    """
    width, height = size
    mask = Image.new('L', size, 0)
   
    for y in range(height):
        for x in range(width):
            value = 0
           
            if pattern_type == "grid":
                # Patrón de cuadrícula
                if x % scale < scale // 2 or y % scale < scale // 2:
                    value = 255
            elif pattern_type == "dots":
                # Patrón de puntos
                dist_to_center = math.sqrt(((x % scale) - scale/2)**2 +
                                        ((y % scale) - scale/2)**2)
                if dist_to_center < scale // 4:
                    value = 255
            elif pattern_type == "stripes":
                # Patrón de rayas
                if (x + y) % scale < scale // 2:
                    value = 255
            elif pattern_type == "waves":
                # Patrón de ondas
                value = int(127.5 + 127.5 * math.sin(x / scale) * math.cos(y / scale))
            else:
                value = 0
           
            mask.putpixel((x, y), value)
   
    return mask
